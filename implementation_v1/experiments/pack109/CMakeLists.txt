cmake_minimum_required(VERSION 3.16)
project(prag_pack109_candidate LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(OpenSSL 3 REQUIRED)
find_package(Threads REQUIRED)
find_package(SEAL 4.1.2 EXACT REQUIRED PATHS "${ROOT}/third_party/SEAL/build/cmake" NO_DEFAULT_PATH)
add_executable(prag_pack109_matrix matrix_probe.cpp bfv_candidate.cpp secure_ops_candidate.cpp)
target_include_directories(prag_pack109_matrix PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include" "${ROOT}/native/include" "${ROOT}/third_party/EzPC/SCI/src")
target_compile_options(prag_pack109_matrix PRIVATE -maes -msse4.1 -mavx -mavx2 -Wno-deprecated-declarations -Wno-ignored-attributes)
target_compile_definitions(prag_pack109_matrix PRIVATE EMP_USE_RANDOM_DEVICE=1 PRAG_NATIVE_VERSION="1.0.9-candidate")
target_link_libraries(prag_pack109_matrix PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/libprag_crypto_v108.a" SEAL::seal OpenSSL::SSL OpenSSL::Crypto Threads::Threads gmpxx gmp)
