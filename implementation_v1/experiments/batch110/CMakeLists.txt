cmake_minimum_required(VERSION 3.16)
project(prag_batch110_probes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${ROOT}/native" native)
# Candidate compiled with the production library's locked dependencies; no
# reconstruction symbol or PRAG_ISOLATED_DEVELOPMENT enters the library.
add_executable(prag_batch110_rms rms_probe.cpp)
target_include_directories(prag_batch110_rms PRIVATE "${ROOT}/compute" "${ROOT}/experiments/batch110_baseline")
target_link_libraries(prag_batch110_rms PRIVATE prag_crypto)
add_executable(prag_batch110_primitives primitive_probe.cpp)
target_link_libraries(prag_batch110_primitives PRIVATE prag_crypto)
add_executable(prag_batch110_attention attention_probe.cpp)
target_include_directories(prag_batch110_attention PRIVATE "${ROOT}/compute" "${ROOT}/experiments/batch110_baseline")
target_link_libraries(prag_batch110_attention PRIVATE prag_crypto)
add_executable(prag_batch110_failure failure_probe.cpp)
target_link_libraries(prag_batch110_failure PRIVATE prag_crypto)

add_executable(prag_batch110_candidate_matrix candidate_matrix_probe.cpp)
target_link_libraries(prag_batch110_candidate_matrix PRIVATE prag_crypto)
