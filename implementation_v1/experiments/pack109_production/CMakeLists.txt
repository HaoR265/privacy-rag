cmake_minimum_required(VERSION 3.16)
project(prag_pack109_production_probes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_subdirectory("${ROOT}/native" native)
foreach(name matrix diagonal token)
 add_executable(prag_pack109_${name} ${name}_probe.cpp)
 target_include_directories(prag_pack109_${name} PRIVATE "${ROOT}/compute")
 target_link_libraries(prag_pack109_${name} PRIVATE prag_crypto)
endforeach()
