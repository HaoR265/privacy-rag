function(prag_locked_source directory expected)
 execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" rev-parse HEAD OUTPUT_VARIABLE actual OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE status)
 if(NOT status EQUAL 0 OR NOT actual STREQUAL expected)
  message(FATAL_ERROR "Locked dependency commit mismatch: ${directory}")
 endif()
 execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${directory}" diff --quiet HEAD RESULT_VARIABLE dirty)
 if(NOT dirty EQUAL 0)
  message(FATAL_ERROR "Tracked dependency source differs from lock: ${directory}")
 endif()
endfunction()

function(prag_library target)
 add_library(${target} STATIC ${PRAG_SOURCES})
 target_include_directories(${target} PUBLIC include "${SCI_ROOT}" PRIVATE gpu/client_phase)
 target_compile_options(${target} PUBLIC -maes -msse4.1 -mavx -mavx2 PRIVATE -Wno-deprecated-declarations -Wno-ignored-attributes)
 target_compile_definitions(${target} PUBLIC EMP_USE_RANDOM_DEVICE=1 PRAG_NATIVE_VERSION="${PROJECT_VERSION}")
 target_link_libraries(${target} PUBLIC SEAL::seal OpenSSL::SSL OpenSSL::Crypto Threads::Threads gmpxx gmp)
 if(PRAG_CLIENT_PHASE_GPU)
  target_compile_definitions(${target} PUBLIC PRAG_CLIENT_PHASE_GPU=1)
  target_link_libraries(${target} PUBLIC prag_client_phase_gpu)
 endif()
 if(PRAG_EXPERIMENTAL_SILU_GPU)
  target_compile_definitions(${target} PUBLIC PRAG_EXPERIMENTAL_SILU_GPU=1)
  target_link_libraries(${target} PUBLIC prag_silu_gpu)
 endif()
 if(PRAG_EXPERIMENTAL_TOKEN_GPU)
  target_compile_definitions(${target} PUBLIC PRAG_EXPERIMENTAL_TOKEN_GPU=1 PRAG_TOKEN_SINGLE_GPU_ROUTE=1 PRAG_TOKEN_PACK_RESIDENT_PAIR=1 PRAG_TOKEN_COORDINATE_PIPELINE=1 PRAG_TOKEN_FULL_DEVICE_RESIDENCY=1 PRAG_TOKEN_LOCAL_RING_GPU=1)
  target_link_libraries(${target} PUBLIC prag_token_gpu)
 endif()
 if(PRAG_EXPERIMENTAL_MATRIX_GPU)
  target_compile_definitions(${target} PUBLIC PRAG_EXPERIMENTAL_MATRIX_GPU=1 PRAG_MATRIX_PENDING_EVALUATE=1)
  target_link_libraries(${target} PUBLIC prag_matrix_gpu)
 endif()
 if(PRAG_EXPERIMENTAL_ENC0_GPU)
  target_compile_definitions(${target} PUBLIC PRAG_ENC0_GPU=1 PRAG_MATRIX_INPUT_ENC_GPU=1)
  target_link_libraries(${target} PUBLIC prag_enc0_gpu)
 endif()
endfunction()
