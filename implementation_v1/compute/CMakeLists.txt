cmake_minimum_required(VERSION 3.16)
project(private_rag_compute LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_subdirectory(../native native)
enable_language(CUDA)
# nvcc uses GCC 13 locally; do not let its implicit search directory shadow
# the C++ compiler's GCC 15 runtime needed by the locked SEAL static library.
list(FILTER CMAKE_CUDA_IMPLICIT_LINK_DIRECTORIES EXCLUDE REGEX "/gcc/[^/]+/[0-9]+$")
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_ARCHITECTURES 75)
add_library(prag_local_cuda STATIC local_shares.cu)
target_include_directories(prag_local_cuda PUBLIC . ../native/include)
add_library(prag_model STATIC model_compile.cpp)
target_include_directories(prag_model PUBLIC .)
target_link_libraries(prag_model PUBLIC prag_crypto prag_local_cuda)
add_executable(prag_worker worker_main.cpp)
target_link_libraries(prag_worker PRIVATE prag_model)
option(PRAG_BUILD_MODEL_DEV "Build public synthetic high-level arithmetic probes" OFF)
if(PRAG_BUILD_MODEL_DEV)
 add_executable(prag_dev_model_probe dev_model_probe.cpp)
 target_include_directories(prag_dev_model_probe PRIVATE .)
 target_link_libraries(prag_dev_model_probe PRIVATE prag_crypto_development)
 add_executable(prag_dev_numeric_operator_probe dev_numeric_operator_probe.cpp)
 target_include_directories(prag_dev_numeric_operator_probe PRIVATE .)
 target_link_libraries(prag_dev_numeric_operator_probe PRIVATE prag_crypto_development)
 add_executable(prag_dev_session_probe dev_session_probe.cpp)
 target_include_directories(prag_dev_session_probe PRIVATE .)
 target_link_libraries(prag_dev_session_probe PRIVATE prag_crypto_development)
 add_executable(prag_dev_numeric_guard_probe dev_numeric_guard_probe.cpp)
 target_include_directories(prag_dev_numeric_guard_probe PRIVATE .)
 target_link_libraries(prag_dev_numeric_guard_probe PRIVATE prag_crypto_development)
 add_executable(prag_dev_linear_registration_probe dev_linear_registration_probe.cpp)
 target_include_directories(prag_dev_linear_registration_probe PRIVATE .)
 target_link_libraries(prag_dev_linear_registration_probe PRIVATE prag_crypto)
endif()
option(PRAG_BUILD_HYBRID_DEV "Build isolated hybrid backend diagnostics" OFF)
if(PRAG_BUILD_HYBRID_DEV)
 add_executable(prag_dev_heartbeat_finish_probe ../native/src/dev_heartbeat_finish_probe.cpp)
 target_link_libraries(prag_dev_heartbeat_finish_probe PRIVATE prag_crypto)

 target_include_directories(prag_token_batch_probe PRIVATE .)
 target_link_libraries(prag_token_batch_probe PRIVATE prag_crypto)

 target_include_directories(prag_full_private_session PRIVATE ../experiments/batch110_baseline)
 target_link_libraries(prag_full_private_session PRIVATE prag_model)
 add_executable(prag_hybrid_local ../experiments/hybrid112/local_probe.cpp)
 target_link_libraries(prag_hybrid_local PRIVATE prag_local_cuda)
